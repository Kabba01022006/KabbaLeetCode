1class Solution {
2    public int countValidPrefixes(String s) {
3        int sum=0;
4        int i=0;
5        int j=1;
6        if(s.length()==1) return 1;
7        sum=s.charAt(i)-'0';
8        int count=1;
9        while(j<s.length()){
10            sum=sum+s.charAt(j)-'0';
11            int len=j-i+1;
12            if(len%2==0){ // if the length is even
13                if(sum==len/2) count++;
14            }
15            if(len%2!=0){ // if the length is odd
16                if(sum==len/2 || sum==(len/2)+1) count++;
17            }
18            j++;
19        }
20        return count;
21    }
