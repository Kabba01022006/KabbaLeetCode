1class Solution {
2    public long maxPairStrength(int[] nums) {
3        long maxStr=0;
4        for(int i=0;i<nums.length;i++){
5            for(int j=i+1;j<nums.length;j++){
6                long str = ((long)nums[i]*(long)nums[j])/(gcd((long)nums[i],(long)nums[j])*gcd((long)nums[i],(long)nums[j]));
7                if(str>maxStr){
8                    maxStr=str;
9                }
10            }
11        }
12        return maxStr;
13    }
14    public long gcd(long a, long b){

17    }
