1class Solution {
2        public long minCost(int m, int n, int[][] penalty) {
3        long[] d = new long[m * n * 2];
4        Arrays.fill(d, Long.MAX_VALUE / 2);
5        PriorityQueue<long[]> q = new PriorityQueue<>(Comparator.comparingLong(a -> a[0]));
6        q.offer(new long[]{1L, 0, 0, 0});
7        int[][] dirs = {{0, 1, 0}, {1, 0, 0}, {0, -1, 1}, {-1, 0, 1}};
8
9        while (!q.isEmpty()) {
10            long[] curr = q.poll();
11            long w = curr[0];
12            int i = (int) curr[1], j = (int) curr[2], p = (int) curr[3];
13
14            if (i == m - 1 && j == n - 1) return w;
15            if (w > d[(i * n + j) * 2 + p]) continue;
16
17            int k = (i * n + j) * 2 + (p ^ 1);
18            if (w + penalty[i][j] < d[k]) {
19                d[k] = w + penalty[i][j];
20                q.offer(new long[]{w + penalty[i][j], i, j, p ^ 1});
21            }
22
23            for (int[] dir : dirs) {
24                int dr = dir[0], dc = dir[1], rp = dir[2];
25                int x = i + dr, y = j + dc;
26                if (0 <= x && x < m && 0 <= y && y < n) {
27                    long w2 = w + (long)(x + 1) * (y + 1) + (p == rp ? 0 : penalty[i][j]);
28                    int next_k = (x * n + y) * 2 + (p ^ 1);
29                    if (w2 < d[next_k]) {
30                        d[next_k] = w2;
31                        q.offer(new long[]{w2, x, y, p ^ 1});
32                    }
33                }
34            }
35        }
36        return -1;
37    }
